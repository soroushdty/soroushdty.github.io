        <script>
        const photoList = [
          '/media/nature-photography/wild-horse-cloud-sea-ridge.jpg',
          '/media/nature-photography/cloud-inversion-mountain-landscape.jpg',
          '/media/nature-photography/forest-stream-sunburst-mist.jpg',
          '/media/nature-photography/crimson-sunset-mountain-ridge.jpg',
          '/media/nature-photography/mountain-valley-village-landscape.jpg',
          '/media/nature-photography/snowy-mountain-range-foothills.jpg',
          '/media/nature-photography/sunset-crepuscular-rays-hills.jpg',
          '/media/nature-photography/ancient-ivy-forest-tree.jpg',
          '/media/nature-photography/pink-twilight-cloud-skyline.jpg',
          '/media/nature-photography/crescent-moon-twilight-horizon.jpg',
          '/media/nature-photography/rugged-mountain-peaks-skyline.jpg',
          '/media/nature-photography/serene-forest-lake-reflection.jpg',
          '/media/nature-photography/high-alpine-meadow-panorama.jpg'
        ];
        let currentPhotoIdx = 0;

        function openPhotoModal(indexOrSrc) {
          if (typeof indexOrSrc === 'number') {
            currentPhotoIdx = indexOrSrc;
          } else {
            const idx = photoList.indexOf(indexOrSrc);
            currentPhotoIdx = idx >= 0 ? idx : 0;
          }
          updateModalState();

          const modal = document.getElementById('photoModal');
          modal.classList.remove('hidden');
          modal.classList.add('flex');
          setTimeout(() => {
            modal.classList.remove('opacity-0');
            modal.classList.add('opacity-100');
          }, 10);
          document.body.style.overflow = 'hidden';
        }

        function updateModalState() {
          const modalImg = document.getElementById('modalImage');
          const counter = document.getElementById('photoCounter');
          modalImg.src = photoList[currentPhotoIdx];
          if (counter) {
            counter.innerText = (currentPhotoIdx + 1) + " / " + photoList.length;
          }
        }

        function prevPhoto(event) {
          if (event) event.stopPropagation();
          currentPhotoIdx = (currentPhotoIdx - 1 + photoList.length) % photoList.length;
          updateModalState();
        }

        function nextPhoto(event) {
          if (event) event.stopPropagation();
          currentPhotoIdx = (currentPhotoIdx + 1) % photoList.length;
          updateModalState();
        }

        function closePhotoModal(event) {
          const modal = document.getElementById('photoModal');
          modal.classList.remove('opacity-100');
          modal.classList.add('opacity-0');
          setTimeout(() => {
            modal.classList.remove('flex');
            modal.classList.add('hidden');
            document.body.style.overflow = '';
          }, 250);
        }

        document.addEventListener('keydown', function(e) {
          const modal = document.getElementById('photoModal');
          if (modal && !modal.classList.contains('hidden')) {
            if (e.key === 'ArrowLeft') {
              prevPhoto();
            } else if (e.key === 'ArrowRight') {
              nextPhoto();
            } else if (e.key === 'Escape') {
              closePhotoModal();
            }
          }
        });
        </script>
